<script>
let yesSize = 1;
let noCount = 0;

function yesClick() {
    document.body.innerHTML = `
    <div style="display:flex;justify-content:center;align-items:center;height:100vh;
    background:black;color:white;text-align:center;">
        <div>
            <h1>YAAAAY LOVE!!! 💖💍</h1>
            <p>You just made me the happiest person alive 🥰</p>
            <h2>I promise to love you forever 💕</h2>
        </div>
    </div>
    `;
}

function noClick() {
    noCount++;
    yesSize += 0.4;
    document.getElementById("yesBtn").style.transform = "scale(" + yesSize + ")";

    const message = document.getElementById("message");
    const emoji = document.getElementById("emoji");

    if(noCount === 1){
        message.innerHTML = "You're breaking my heart Love 💔";
        emoji.innerHTML = "<div class='broken'>💔</div>";
    }
    else if(noCount === 2){
        message.innerHTML = "It really hurts... I thought I was special to you 🥺💔";
        emoji.innerHTML = "<div class='broken'>💔</div>";
    }
    else if(noCount >= 3){
        message.innerHTML = "Okay... I can't handle this anymore 😭";
        emoji.innerHTML = "<div style='font-size:70px;'>😭</div>";
    }
}
</script>
